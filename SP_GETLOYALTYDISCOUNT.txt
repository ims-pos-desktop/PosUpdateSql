CREATE OR ALTER   PROCEDURE [dbo].[SP_GETLOYALTYDISCOUNT] 
--DECLARE
	@CODE VARCHAR(20) , 
	@DIVISION CHAR(3) = ''

AS

/*
SELECT @code=N'M1602' , @DIVISION=N'BSP'
--*/

SET NOCOUNT ON

SELECT a.DisID schemeID , i.MCODE , C.disrate SchemeDisRate , C.disamount SchemeDisAmount , a.comboid , a.schemename , a.priority , 1 MinQty , a.SchemeType , a.IntervalLimitBy , a.LimitIntervalReset , a.IntervalLimitAmount , a.IntervalLimitQty , a.IsActive
FROM MENUITEM I
INNER JOIN Discount_SchemeDiscount C ON I.MCODE = C.Mcode OR I.PARENT = C.Parent OR I.MGROUP = C.MGroup
INNER JOIN Discount_Rate A ON C.DisID = A.DisID
INNER JOIN vwSchemeSchedule B ON A.ScheduleID = B.DisID
INNER JOIN MEMBERSCHEME MS ON MS.SCHEMEID = A.SchemeType
WHERE I.MCODE = @CODE AND ISNULL(I.DISMODE , 'DISCOUNTABLE') = 'DISCOUNTABLE'
AND (@DIVISION = '%' OR isnull(a.divisions,'') = '' OR @DIVISION in (select * from split(a.divisions , ',')))
AND A.IsActive = 1
ORDER BY A.Priority
